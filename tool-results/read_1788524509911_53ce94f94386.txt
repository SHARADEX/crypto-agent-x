     3→// LifecycleTab — the opportunity state-machine flowchart (Phase-3 STYLING-1).
     4→//
     5→// Visualizes the 14 OpportunityStatus values as a 3-row flowchart so operators
     6→// can see at a glance where opportunities are piling up. Each node is a small
     7→// card showing the status name, a Lucide icon, the count of opportunities
     8→// currently in that status, and a colored border that matches the status'
     9→// semantic color (sourced from `statusColor`).
    10→//
    11→// Layout:
    12→//   Row 1 (Discovery):   discovered → researching → verified → rejected
    13→//   Row 2 (Planning):     queued → planning → approved → executing → executed
    14→//   Row 3 (Post-submit):  submitted → awaiting_payment → needs_improvement → paid / failed
    15→//
    16→// Interactions:
    17→//   - Clicking a node calls `onSelectStatus(status)` — the parent switches
    18→//     to the Opportunities tab pre-filtered to that status.
    19→//   - Nodes for non-terminal statuses whose oldest opportunity is > 24h
    20→//     old get an amber pulse ring (the "stuck" indicator).
    21→//   - A sticky callout at the top lists every stuck status with a one-click
    22→//     "jump to filtered list" CTA.
    23→//
    24→// Data source: GET /api/analytics/lifecycle →
    25→//   { statuses: [{ status, count, oldestUpdatedAt }] }
    26→//
    27→// No heavy graph library is used — the flowchart is pure Tailwind divs + a
    28→// handful of SVG arrows with `marker-end` chevrons.
    29→
    30→import * as React from "react";
    31→import { useQuery } from "@tanstack/react-query";
    32→import {
    33→  Area,
    34→  AreaChart,
    35→  ResponsiveContainer,
    36→  Tooltip as RTooltip,

    40→import {

    42→  CardContent,
    43→  CardDescription,

    46→} from "@/components/ui/card";

    57→import { statusIcon } from "./lib/status-badge";

    60→// ---------------------------------------------------------------------------
    61→// Layout: the 3 rows of the state machine + terminal-cluster definitions.
    62→// ---------------------------------------------------------------------------
    63→
    64→/** Stages (rows) of the opportunity lifecycle flowchart. */
    65→const FLOW_ROWS: { label: string; hint: string; statuses: string[] }[] = [
    66→  {
    67→    label: "Discovery",
    68→    hint: "Scanner → research → verification",
    69→    statuses: ["discovered", "researching", "verified", "rejected"],
    70→  },
    71→  {
    72→    label: "Planning & Execution",
    73→    hint: "Queue → plan → approve → execute",
    74→    statuses: ["queued", "planning", "approved", "executing", "executed"],
    75→  },
    76→  {
    77→    label: "Post-submit",
    78→    hint: "PR merged → payment → done",
    79→    statuses: [
    80→      "submitted",
    81→      "awaiting_payment",
    82→      "needs_improvement",
    83→      "paid",
    84→      "failed",

    90→const TERMINAL_STATUSES = new Set([
    91→  "rejected",
    92→  "paid",
    93→  "failed",
    94→  "skipped",
    95→  "cancelled",
    96→]);
    97→
    98→/** The 24h "stuck" threshold in ms. */
    99→const STUCK_THRESHOLD_MS = 24 * 60 * 60 * 1000;

   104→
   105→export function LifecycleTab({
   106→  onSelectStatus,
   107→}: {
   108→  /**
   109→   * Called when the user clicks a status node. The parent typically switches
   110→   * to the Opportunities tab pre-filtered to that status.
   111→   */
   112→  onSelectStatus: (status: string) => void;
   113→}) {
   114→  // Phase-3 dev-review #3 (priority #4): toggle between the fast midpoint
   115→  // approximation (default) and the true arithmetic mean (fetches every
   116→  // row's updatedAt — more accurate for skewed distributions, but O(n)).
   117→  const [trueAverage, setTrueAverage] = React.useState(false);
   118→
   119→  const { data, isLoading, error } = useQuery({
   120→    queryKey: ["analytics", "lifecycle", { trueAverage }],
   121→    queryFn: () => api.analytics.lifecycle({ trueAverage }),
   122→    refetchInterval: 15_000,
   123→  });
   124→
   125→  // Phase-3 dev-review #3 (priority #5): historical trend chart. Fetches
   126→  // the last 7 days of lifecycle snapshots (one row per status per hour).
   127→  // The chart renders one area per status showing how the count trended.
   128→  const { data: snapshotData, isLoading: snapshotsLoading } = useQuery({
   129→    queryKey: ["analytics", "lifecycle", "snapshots", "7d"],
   130→    queryFn: () =>
   131→      fetch("/api/analytics/lifecycle/snapshot?days=7").then((r) => r.json()),
   132→    refetchInterval: 60_000, // refresh every minute
   133→  });
   134→
   135→  // Build a chart-friendly data shape: one row per bucket, with a column
   136→  // per status (so the area chart can stack them).
   137→  const trendChartData = React.useMemo(() => {
   138→    const snapshots: Array<{
   139→      bucket: string;
   140→      status: string;
   141→      count: number;
   142→      avgHoursInStatus: number;
   143→      maxHoursInStatus: number;
   144→      createdAt: string;
   145→    }> = snapshotData?.snapshots ?? [];
   146→    if (snapshots.length === 0) return [];
   147→    // Group by bucket.
   148→    const byBucket = new Map<string, Record<string, number | string>>();
   149→    for (const s of snapshots) {
   150→      const entry = byBucket.get(s.bucket) ?? { bucket: s.bucket };
   151→      entry[s.status] = s.count;
   152→      byBucket.set(s.bucket, entry);
   153→    }
   154→    // Sort by bucket ascending + return as array.
   155→    return Array.from(byBucket.values()).sort((a, b) =>
   156→      String(a.bucket).localeCompare(String(b.bucket))
   157→    );
   158→  }, [snapshotData]);
   159→
   160→  // The set of statuses present in the snapshots (for the chart legend).
   161→  const trendStatuses = React.useMemo(() => {
   162→    const set = new Set<string>();
   163→    for (const row of trendChartData) {
   164→      for (const key of Object.keys(row)) {
   165→        if (key !== "bucket") set.add(key);
   166→      }
   167→    }
   168→    return Array.from(set).sort();
   169→  }, [trendChartData]);
   170→
   171→  // Build a lookup map: status → { count, oldestUpdatedAt, avgHoursInStatus, maxHoursInStatus }.
   172→  const byStatus = React.useMemo(() => {
   173→    const m = new Map<
   174→      string,
   175→      {
   176→        count: number;
   177→        oldestUpdatedAt: string | null;
   178→        avgHoursInStatus?: number;
   179→        maxHoursInStatus?: number;
   180→      }
   181→    >();
   182→    for (const row of data?.statuses ?? []) {
   183→      m.set(row.status, {
   184→        count: row.count,
   185→        oldestUpdatedAt: row.oldestUpdatedAt,
   186→        avgHoursInStatus: row.avgHoursInStatus,
   187→        maxHoursInStatus: row.maxHoursInStatus,
   188→      });
   189→    }
   190→    // Ensure every status in FLOW_ROWS has an entry (so empty nodes render).
   191→    for (const row of FLOW_ROWS) {
   192→      for (const s of row.statuses) {
   193→        if (!m.has(s)) {
   194→          m.set(s, { count: 0, oldestUpdatedAt: null });
   195→        }
   196→      }
   197→    }
   198→    return m;
   199→  }, [data]);
   200→
   201→  // Compute the global max avgHoursInStatus across all statuses — used to
   202→  // scale the per-node sparkline bars so they're comparable across nodes.
   203→  const maxAvgHoursForScale = React.useMemo(() => {
   204→    let max = 0;
   205→    for (const row of data?.statuses ?? []) {
   206→      if (typeof row.avgHoursInStatus === "number" && row.avgHoursInStatus > max) {
   207→        max = row.avgHoursInStatus;
   208→      }
   209→    }
   210→    // Floor at 24h so the "stuck threshold" is visually anchored.
   211→    return Math.max(max, 24);
   212→  }, [data]);
   213→
   214→  // Compute the set of "stuck" statuses (non-terminal + oldest > 24h).
   215→  const stuckStatuses = React.useMemo(() => {
   216→    if (!data?.statuses) return [];
   217→    const now = Date.now();
   218→    const stuck: { status: string; count: number; ageHours: number }[] = [];
   219→    for (const row of data.statuses) {
   220→      if (TERMINAL_STATUSES.has(row.status)) continue;
   221→      if (row.count === 0) continue;
   222→      if (!row.oldestUpdatedAt) continue;
   223→      const oldest = Date.parse(row.oldestUpdatedAt);
   224→      if (!Number.isFinite(oldest)) continue;
   225→      const age = now - oldest;
   226→      if (age > STUCK_THRESHOLD_MS) {
   227→        stuck.push({
   228→          status: row.status,
   229→          count: row.count,
   230→          ageHours: Math.floor(age / (60 * 60 * 1000)),
   231→        });
   232→      }
   233→    }
   234→    return stuck.sort((a, b) => b.ageHours - a.ageHours);
   235→  }, [data]);
   236→
   237→  if (error) {
   238→    return (
   239→      <div className="rounded-md border border-red-500/30 bg-red-500/10 p-4 text-sm text-red-700 dark:text-red-300">
   240→        Failed to load lifecycle analytics: {String(error)}
   241→      </div>
   242→    );
   243→  }
   244→
   245→  return (
   246→    <div className="space-y-4">
   247→      {/* Header + Stuck callout */}
   248→      <Card className="border-border/60">
   249→        <CardHeader className="pb-3">
   250→          <div className="flex items-center justify-between gap-2">
   251→            <CardTitle className="flex items-center gap-2 text-base">
   252→              <RotateCw className="size-4 text-teal-500" />
   253→              Opportunity Lifecycle
   254→            </CardTitle>
   255→            {/* Phase-3 dev-review #3 (priority #4): true-average toggle. */}
   256→            <label className="flex cursor-pointer items-center gap-1.5 text-[11px] text-muted-foreground">
   257→              <input
   258→                type="checkbox"
   259→                className="size-3 cursor-pointer accent-teal-500"
   260→                checked={trueAverage}
   261→                onChange={(e) => setTrueAverage(e.target.checked)}
   262→                title="When ON, the avg hours are the true arithmetic mean (more accurate for skewed distributions, but slower — fetches every opportunity). When OFF, the avg is the midpoint between oldest + newest (fast approximation)."
   263→              />
   264→              True average
   265→              {data?.trueAverage ? (
   266→                <span className="rounded-sm bg-teal-500/15 px-1 text-[9px] font-medium text-teal-700 dark:text-teal-300">
   267→                  exact
   268→                </span>
   269→              ) : (
   270→                <span className="rounded-sm bg-muted px-1 text-[9px] text-muted-foreground">
   271→                  approx
   272→                </span>
   273→              )}
   274→            </label>
   275→          </div>
   276→          <CardDescription>
   277→            The opportunity state machine — 14 statuses grouped into 3 stages.
   278→            Click any node to filter the Opportunities tab to that status.
   279→          </CardDescription>
   280→        </CardHeader>
   281→        {stuckStatuses.length > 0 && (
   282→          <CardContent className="pt-0">
   283→            <div className="flex flex-wrap items-start gap-3 rounded-lg border border-amber-500/40 bg-amber-500/10 p-3 text-amber-800 dark:text-amber-200">
   284→              <AlertTriangle className="mt-0.5 size-4 shrink-0 animate-pulse text-amber-500" />
   285→              <div className="min-w-0 flex-1">
   286→                <div className="text-sm font-medium">
   287→                  {stuckStatuses.length} status
   288→                  {stuckStatuses.length === 1 ? "" : "es"} stuck for more than
   289→                  24 hours
   290→                </div>
   291→                <div className="mt-1 flex flex-wrap gap-1.5">
   292→                  {stuckStatuses.map((s) => (
   293→                    <button
   294→                      key={s.status}
   295→                      type="button"
   296→                      onClick={() => onSelectStatus(s.status)}
   297→                      className="inline-flex items-center gap-1 rounded-full border border-amber-500/40 bg-amber-500/20 px-2 py-0.5 text-[11px] font-medium capitalize transition-colors hover:bg-amber-500/30"
   298→                    >
   299→                      {s.status.replace(/_/g, " ")}
   300→                      <span className="tabular-nums opacity-70">
   301→                        · {s.count} for {s.ageHours}h
   302→                      </span>
   303→                    </button>
   304→                  ))}
   305→                </div>
   306→              </div>
   307→            </div>
   308→          </CardContent>
   309→        )}
   310→      </Card>
   311→
   312→      {/* Flowchart */}
   313→      <div className="space-y-3">
   314→        {FLOW_ROWS.map((row, rowIdx) => (
   315→          <Card
   316→            key={row.label}
   317→            className="overflow-hidden border-border/60"
   318→          >
   319→            <CardHeader className="pb-2">
   320→              <div className="flex items-baseline justify-between gap-2">
   321→                <CardTitle className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   322→                  Stage {rowIdx + 1} · {row.label}
   323→                </CardTitle>
   324→                <span className="text-[10px] text-muted-foreground">
   325→                  {row.hint}
   326→                </span>
   327→              </div>
   328→            </CardHeader>
   329→            <CardContent>
   330→              {isLoading ? (
   331→                <Skeleton className="h-24 w-full" />
   332→              ) : (
   333→                <div className="flex flex-wrap items-stretch gap-2 sm:flex-nowrap">
   334→                  {row.statuses.map((status, i) => (
   335→                    <React.Fragment key={status}>
   336→                      <StatusNode
   337→                        status={status}
   338→                        count={byStatus.get(status)?.count ?? 0}
   339→                        oldestUpdatedAt={
   340→                          byStatus.get(status)?.oldestUpdatedAt ?? null
   341→                        }
   342→                        avgHoursInStatus={
   343→                          byStatus.get(status)?.avgHoursInStatus
   344→                        }
   345→                        maxHoursInStatus={
   346→                          byStatus.get(status)?.maxHoursInStatus
   347→                        }
   348→                        maxAvgHoursForScale={maxAvgHoursForScale}
   349→                        onClick={() => onSelectStatus(status)}
   350→                      />
   351→                      {i < row.statuses.length - 1 && <NodeArrow />}
   352→                    </React.Fragment>
   353→                  ))}
   354→                </div>
   355→              )}
   356→            </CardContent>
   357→          </Card>
   358→        ))}
   359→      </div>
   360→
   361→      {/* Legend */}
   362→      <Card className="border-border/60">
   363→        <CardHeader className="pb-2">
   364→          <CardTitle className="flex items-center gap-2 text-sm">
   365→            <Info className="size-4 text-muted-foreground" />
   366→            Legend
   367→          </CardTitle>
   368→        </CardHeader>
   369→        <CardContent>
   370→          <div className="grid grid-cols-2 gap-2 text-xs sm:grid-cols-3 lg:grid-cols-4">
   371→            <LegendItem
   372→              label="Discovery"
   373→              swatchClass="bg-slate-500"
   374→              hint="discovered, researching"
   375→            />
   376→            <LegendItem
   377→              label="Verification"
   378→              swatchClass="bg-cyan-500"
   379→              hint="researching"
   380→            />
   381→            <LegendItem
   382→              label="Verified / Queued"
   383→              swatchClass="bg-emerald-500"
   384→              hint="verified, queued, approved"
   385→            />
   386→            <LegendItem
   387→              label="Active Work"
   388→              swatchClass="bg-teal-500"
   389→              hint="planning, executing"
   390→            />
   391→            <LegendItem
   392→              label="Awaiting"
   393→              swatchClass="bg-amber-500"
   394→              hint="awaiting_payment"
   395→            />
   396→            <LegendItem
   397→              label="Submitted PR"
   398→              swatchClass="bg-orange-500"
   399→              hint="submitted"
   400→            />
   401→            <LegendItem
   402→              label="Needs Work"
   403→              swatchClass="bg-rose-500"
   404→              hint="needs_improvement"
   405→            />
   406→            <LegendItem
   407→              label="Terminal"
   408→              swatchClass="bg-red-500"
   409→              hint="rejected, failed"
   410→            />
   411→          </div>
   412→          <div className="mt-3 flex items-center gap-2 text-[11px] text-muted-foreground">
   413→            <span className="inline-flex h-3 w-3 animate-pulse rounded-full border-2 border-amber-500" />
   414→            Amber pulse = a status with at least one opportunity older than 24h.
   415→            Click to triage.
   416→          </div>
   417→        </CardContent>
   418→      </Card>
   419→
   420→      {/* Phase-3 dev-review #3 (priority #5): 7-day trend chart. Shows how
   421→          the count per status has changed over the last 7 days so the
   422→          operator can see whether bottlenecks are improving or worsening. */}
   423→      <Card className="border-border/60">
   424→        <CardHeader className="pb-2">
   425→          <CardTitle className="flex items-center gap-2 text-sm">
   426→            <TrendingUp className="size-4 text-teal-500" />
   427→            7-day trend
   428→          </CardTitle>
   429→          <CardDescription>
   430→            Opportunity count per status over the last 7 days. Stacked areas
   431→            show where the pipeline is backing up. Data is snapshotted hourly
   432→            (POST /api/analytics/lifecycle/snapshot).
   433→          </CardDescription>
   434→        </CardHeader>
   435→        <CardContent>
   436→          {snapshotsLoading ? (
   437→            <Skeleton className="h-48 w-full" />
   438→          ) : trendChartData.length === 0 ? (
   439→            <div className="flex h-48 items-center justify-center rounded-md border border-dashed border-border/60 text-xs text-muted-foreground">
   440→              No snapshots yet. Run{" "}
   441→              <code className="mx-1 rounded bg-muted px-1 py-0.5 text-[10px]">
   442→                curl -X POST /api/analytics/lifecycle/snapshot
   443→              </code>{" "}
   444→              to seed the trend data.
   445→            </div>
   446→          ) : (
   447→            <ResponsiveContainer width="100%" height={220}>
   448→              <AreaChart data={trendChartData}>
   449→                <defs>
   450→                  {trendStatuses.map((status) => {
   451→                    const c = statusColor(status);
   452→                    return (
   453→                      <linearGradient
   454→                        key={status}
   455→                        id={`grad-${status}`}
   456→                        x1="0"
   457→                        y1="0"
   458→                        x2="0"
   459→                        y2="1"
   460→                      >
   461→                        <stop
   462→                          offset="5%"
   463→                          stopColor="currentColor"
   464→                          stopOpacity={0.3}
   465→                        />
   466→                        <stop
   467→                          offset="95%"
   468→                          stopColor="currentColor"
   469→                          stopOpacity={0}
   470→                        />
   471→                      </linearGradient>
   472→                    );
   473→                  })}
   474→                </defs>
   475→                <XAxis
   476→                  dataKey="bucket"
   477→                  tick={{ fontSize: 9 }}
   478→                  tickFormatter={(b: string) => {
   479→                    // Phase-3 DEV-REVIEW-6 (#4): format the bucket string
   480→                    // (YYYY-MM-DDTHH) as a friendly "Mon DD" label.
   481→                    // The bucket is UTC, so we parse it as such + format
   482→                    // with the user's locale for readability.
   483→                    const date = new Date(b + ":00:00.000Z");
   484→                    if (isNaN(date.getTime())) return b.slice(0, 10);
   485→                    return date.toLocaleDateString(undefined, {
   486→                      month: "short",
   487→                      day: "numeric",
   488→                    });
   489→                  }}
   490→                  interval="preserveStartEnd"
   491→                />
   492→                <YAxis
   493→                  tick={{ fontSize: 9 }}
   494→                  allowDecimals={false}
   495→                />
   496→                <RTooltip
   497→                  contentStyle={{
   498→                    fontSize: 11,
   499→                    background: "rgba(0,0,0,0.85)",
   500→                    border: "none",
   501→                    borderRadius: 4,
   502→                    color: "#fff",
   503→                  }}
   504→                  labelStyle={{ fontSize: 10, opacity: 0.7 }}
   505→                  // Phase-3 DEV-REVIEW-7 (#3): format the tooltip label
   506→                  // (the bucket string YYYY-MM-DDTHH) as a friendly
   507→                  // "Aug 23, 14:00" matching the X-axis style.
   508→                  labelFormatter={(b: string) => {
   509→                    const date = new Date(b + ":00:00.000Z");
   510→                    if (isNaN(date.getTime())) return b;
   511→                    return date.toLocaleString(undefined, {
   512→                      month: "short",
   513→                      day: "numeric",
   514→                      hour: "2-digit",
   515→                      minute: "2-digit",
   516→                      hour12: false,
   517→                    });
   518→                  }}
   519→                />
   520→                {trendStatuses.map((status) => {
   521→                  const c = statusColor(status);
   522→                  return (
   523→                    <Area
   524→                      key={status}
   525→                      type="monotone"
   526→                      dataKey={status}
   527→                      stackId="1"
   528→                      stroke={c.text}
   529→                      fill={`url(#grad-${status})`}
   530→                      fillOpacity={1}
   531→                    />
   532→                  );
   533→                })}
   534→              </AreaChart>
   535→            </ResponsiveContainer>
   536→          )}
   537→          {trendStatuses.length > 0 ? (
   538→            <div className="mt-2 flex flex-wrap gap-2">
   539→              {trendStatuses.map((status) => {
   540→                const c = statusColor(status);
   541→                return (
   542→                  <div
   543→                    key={status}
   544→                    className="flex items-center gap-1 text-[10px]"
   545→                  >
   546→                    <span
   547→                      className={cn("size-2 rounded-full", c.bg)}
   548→                      style={{ color: "currentColor" }}
   549→                    />
   550→                    <span className="capitalize">{status.replace(/_/g, " ")}</span>
   551→                  </div>
   552→                );
   553→              })}
   554→            </div>
   555→          ) : null}
   556→        </CardContent>
   557→      </Card>
   558→    </div>
   559→  );
   560→}

   562→// ---------------------------------------------------------------------------
   563→// Sub-components
   564→// ---------------------------------------------------------------------------
   565→
   566→/**
   567→ * A single status node in the flowchart. Renders as a small card with:
   568→ *   - colored top + left border (matches the status' semantic color)
   569→ *   - the Lucide icon for that status
   570→ *   - the status name (humanized)
   571→ *   - the count of opportunities currently in that status (tabular-nums)
   572→ *   - an amber pulse ring when the status is non-terminal + oldest > 24h
   573→ *
   574→ * The whole card is a button so it's keyboard-accessible.
   575→ */
   576→function StatusNode({
   577→  status,
   578→  count,
   579→  oldestUpdatedAt,
   580→  avgHoursInStatus,
   581→  maxHoursInStatus,
   582→  maxAvgHoursForScale,
   583→  onClick,
   584→}: {
   585→  status: string;
   586→  count: number;
   587→  oldestUpdatedAt: string | null;
   588→  avgHoursInStatus?: number;
   589→  maxHoursInStatus?: number;
   590→  maxAvgHoursForScale: number;
   591→  onClick: () => void;
   592→}) {
   593→  const c = statusColor(status);
   594→  const Icon = statusIcon(status);
   595→  const isTerminal = TERMINAL_STATUSES.has(status);
   596→  const isStuck = React.useMemo(() => {
   597→    if (isTerminal || count === 0 || !oldestUpdatedAt) return false;
   598→    const age = Date.now() - Date.parse(oldestUpdatedAt);
   599→    return Number.isFinite(age) && age > STUCK_THRESHOLD_MS;
   600→  }, [isTerminal, count, oldestUpdatedAt]);
   601→
   602→  // Sparkline bar: shows avgHoursInStatus relative to maxAvgHoursForScale.
   603→  // Renders a horizontal bar with two segments — the "healthy" portion (< 24h)
   604→  // in the status's accent color, and the "stuck" portion (> 24h) in amber.
   605→  // Only rendered when count > 0 + avgHoursInStatus is a finite number.
   606→  const showSparkline =
   607→    count > 0 &&
   608→    typeof avgHoursInStatus === "number" &&
   609→    Number.isFinite(avgHoursInStatus);
   610→  const sparklinePct =
   611→    typeof avgHoursInStatus === "number" && maxAvgHoursForScale > 0
   612→      ? Math.min(100, (avgHoursInStatus / maxAvgHoursForScale) * 100)
   613→      : 0;
   614→  const stuckPct =
   615→    typeof avgHoursInStatus === "number" && avgHoursInStatus > 24
   616→      ? Math.min(100, ((avgHoursInStatus - 24) / maxAvgHoursForScale) * 100)
   617→      : 0;
   618→  const healthyPct = Math.max(0, sparklinePct - stuckPct);
   619→
   620→  return (
   621→    <button
   622→      type="button"
   623→      onClick={onClick}
   624→      className={cn(
   625→        "group relative flex min-w-[120px] flex-1 flex-col gap-1.5 rounded-lg border bg-card/40 p-3 text-left transition-all hover:bg-card/80 hover:shadow-md focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500/50",
   626→        c.border,
   627→        c.borderLeft,
   628→        isStuck &&
   629→          "ring-2 ring-amber-500/70 ring-offset-1 ring-offset-background animate-pulse"
   630→      )}
   631→      aria-label={`Filter opportunities by status: ${status} (${count} current${typeof avgHoursInStatus === "number" ? `, avg ${avgHoursInStatus.toFixed(1)}h in status` : ""})`}
   632→    >
   633→      <div className="flex items-center justify-between gap-1">
   634→        <span
   635→          className={cn(
   636→            "flex size-7 items-center justify-center rounded-full",
   637→            c.bg,
   638→            c.text
   639→          )}
   640→        >
   641→          <Icon
   642→            className={cn(
   643→              "size-3.5",
   644→              (status === "executing" || status === "running") &&
   645→                "animate-spin"
   646→            )}
   647→            aria-hidden
   648→          />
   649→        </span>
   650→        <span
   651→          className={cn(
   652→            "text-base font-semibold tracking-tight tabular-nums",
   653→            c.text
   654→          )}
   655→        >
   656→          {count}
   657→        </span>
   658→      </div>
   659→      <div className="text-[11px] font-medium capitalize leading-tight text-foreground">
   660→        {status.replace(/_/g, " ")}
   661→      </div>
   662→      {showSparkline ? (
   663→        <div className="mt-0.5 space-y-0.5">
   664→          {/* Sparkline bar — healthy portion in accent color, stuck portion in amber. */}
   665→          <div
   666→            className="relative h-1.5 w-full overflow-hidden rounded-full bg-muted/60"
   667→            role="img"
   668→            aria-label={`Average ${avgHoursInStatus?.toFixed(1)} hours in this status (max ${maxHoursInStatus?.toFixed(1)}h)`}
   669→          >
   670→            <div
   671→              className={cn("absolute inset-y-0 left-0 rounded-full", c.bg, c.text)}
   672→              style={{ width: `${healthyPct}%` }}
   673→            />
   674→            <div
   675→              className="absolute inset-y-0 rounded-full bg-amber-500/70"
   676→              style={{
   677→                left: `${healthyPct}%`,
   678→                width: `${stuckPct}%`,
   679→              }}
   680→            />
   681→          </div>
   682→          <div className="flex items-center justify-between text-[9px] text-muted-foreground tabular-nums">
   683→            <span>avg {avgHoursInStatus?.toFixed(1)}h</span>
   684→            <span>max {maxHoursInStatus?.toFixed(1)}h</span>
   685→          </div>
   686→        </div>
   687→      ) : isTerminal ? (
   688→        <div className="text-[9px] uppercase tracking-wide text-muted-foreground">
   689→          terminal
   690→        </div>
   691→      ) : null}
   692→    </button>
   693→  );
   694→}
